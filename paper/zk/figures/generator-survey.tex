\ior_open:Nn \l_survey_input_ior {../generated/T-generators.tex}
\ior_str_map_inline:Nn \l_survey_input_ior
  {
    \regex_split:nnN {\s \& \s} {#1} \l_tmpa_seq
    \int_compare:nNnT {\seq_count:N \l_tmpa_seq} > {3}
      {
        \str_set:Nx \l_tmpa_str {\seq_item:Nn \l_tmpa_seq {4}}
        \str_if_eq:VnF \l_tmpa_str {behaviour}
          {
            \prop_get:NVNF \l_survey_counts_prop \l_tmpa_str \l_tmpa_tl
              {\tl_set:Nn \l_tmpa_tl {0}}
            \prop_put:NVx \l_survey_counts_prop \l_tmpa_str
              {\int_eval:n {\l_tmpa_tl + 1}}
          }
      }
  }
\ior_close:N \l_survey_input_ior

      {\PackageError{generator-survey}{Missing~survey~category:~#1}
        {Review~the~behaviour~column~of~T-generators.tex.}}
  }
